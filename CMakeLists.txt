cmake_minimum_required(VERSION 3.16)
project(dfam_model CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(dfam INTERFACE)
target_include_directories(dfam INTERFACE include)
add_executable(dfam_render tools/render.cpp)
target_link_libraries(dfam_render dfam)
add_executable(dfam_tests tests/test_dfam.cpp)
target_link_libraries(dfam_tests dfam)
enable_testing()
add_test(NAME dfam_tests COMMAND dfam_tests)
